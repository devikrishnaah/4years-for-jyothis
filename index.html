import tkinter as tk

notes = [
    ("Year 1 💛", "commit 1:\ngit init\nliked you ❤"),
    ("Year 2 💖", "commit 2:\nfeat: adventures\nloved you ❤"),
    ("Year 3 💚", "commit 3:\nfix: stronger\nneeded you ❤"),
    ("Year 4 💜", "commit 4:\nrelease v4.0\nStill my favorite?\nWill you be mine\nfor 4 more? ❤")
]

colors = ["#FFF9C4", "#F8BBD0", "#C8E6C9", "#E1BEE7"]

for i, (title, text) in enumerate(notes):
    win = tk.Tk()
    win.title(title)
    win.geometry(f"220x180+{100+i*250}+100")
    win.configure(bg=colors[i])
    tk.Label(win, text=text, bg=colors[i], font=("Consolas", 11), justify="left").pack(pady=20, padx=10)
    win.attributes("-topmost", True) # always on top like real sticky

tk.mainloop()
